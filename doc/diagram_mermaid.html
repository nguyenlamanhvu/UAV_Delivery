Se3Controller Architecture

flowchart TD
    %% Inputs
    State[LCM: quadrotor_state
Position, Velocity, Rotation]
    Setpoint[LCM: quadrotor_setpoint
Mode, Target Pos/Vel/Yaw]

    %% Main Controller Block
    subgraph Controller [Drake LeafSystem: Se3Controller]
        
        %% Continuous States
        subgraph States [Continuous States / Integrators]
            Integrator((Velocity
Integrator))
            NDOB((Nonlinear
Disturbance
Observer))
        end

        %% Algebraic Core
        subgraph Logic [Control Logic / CalcCommand]
            Router{Mode Router
Pos / Vel / Direct}
            Drag[Aerodynamic
Drag Feedforward]
            ReqForce(Required Force Equation)
            S2S1[S2 x S1 Decoupled
Yaw Controller]
            Prioritizer[Hierarchical Mixer
1. Roll/Pitch 2. Thrust 3. Yaw]
        end

        %% Connections
        State --> NDOB
        State --> Integrator
        State --> Router
        State --> Drag

        Setpoint --> Router
        
        Router -- "v_cmd" --> Integrator
        Router -- "v_cmd" --> ReqForce
        Integrator -- "I_v" --> ReqForce
        NDOB -- "F_dist" --> ReqForce
        Drag -- "F_drag" --> ReqForce
        
        ReqForce --> S2S1
        S2S1 -- "Desired Torques & Thrust" --> Prioritizer
        
        %% Anti-windup feedback
        Prioritizer -. "Anti-Windup Freeze" .-> Integrator
    end

    %% Outputs
    Command[LCM: quadrotor_command
Rotor 1-4 Inputs]
    Prioritizer --> Command
